# AI Narrative Layer Implementation Plan

> **For Claude:** REQUIRED SUB-SKILL: Use superpowers:executing-plans to implement this plan task-by-task.

**Goal:** Build thin AI narrative wrappers in `ai/` that convert structured engine output into human-readable financial insights using Claude Haiku.

**Architecture:** Each insight function has a matching narrator that accepts typed engine output, builds a JSON context string, sends it to Claude Haiku with a locked system prompt, and returns a narrative string. No financial math in `ai/`. All numbers pre-computed by `engine/insights/`.

**Tech Stack:** `@anthropic-ai/sdk` (TypeScript), Claude Haiku (`claude-haiku-4-5`), Next.js API routes

### Task 4: Create Context Builders

**Files:**
- Create: `ai/context-builders.ts`

Context builders are pure functions that convert typed engine output into a JSON string suitable for the LLM user message. They format amounts into human-readable form (major units) so the LLM has clean data to narrate.

**Step 1: Create ai/context-builders.ts**

```typescript
/**
 * Context builders — convert engine output to JSON strings for LLM consumption.
 *
 * Pure functions. No DB, no AI calls, no side effects.
 * Each builder produces a structured JSON string that the LLM narrates.
 */
import type { SpendingTrend } from "../engine/insights/spending-trends";
import type { SpendingAnomaly } from "../engine/insights/anomaly-detection";
import type { DetectedSubscription } from "../engine/insights/subscription-detection";
import type { AffordabilityResult } from "../engine/insights/affordability";
import type { HealthScoreResult } from "../engine/insights/health-score";

function toMajor(minor: number, decimals = 2): string {
  return (minor / Math.pow(10, decimals)).toFixed(decimals);
}

export function buildSpendingTrendsContext(
  trends: readonly SpendingTrend[],
  currency: string
): string {
  const items = trends.map((t) => ({
    category: t.categoryKey,
    current: `${currency} ${toMajor(t.currentMinor)}`,
    baselineAvg: `${currency} ${toMajor(t.baselineAvgMinor)}`,
    percentChange: t.percentChange !== null ? `${t.percentChange.toFixed(1)}%` : "N/A (new category)",
  }));
  return JSON.stringify({ type: "spending_trends", currency, trends: items }, null, 2);
}

export function buildAnomalyContext(
  anomalies: readonly SpendingAnomaly[],
  currency: string
): string {
  const items = anomalies.map((a) => ({
    category: a.categoryKey,
    current: `${currency} ${toMajor(a.currentMinor)}`,
    baselineAvg: `${currency} ${toMajor(a.baselineAvgMinor)}`,
    deviation: `${a.deviationPercent.toFixed(1)}%`,
    direction: a.direction,
  }));
  return JSON.stringify({ type: "spending_anomalies", currency, anomalies: items }, null, 2);
}

export function buildSubscriptionContext(
  subscriptions: readonly DetectedSubscription[],
  currency: string
): string {
  const items = subscriptions.map((s) => ({
    merchant: s.merchantToken,
    averageAmount: `${currency} ${toMajor(Math.abs(s.averageAmountMinor))}`,
    occurrences: s.occurrences,
    latestDate: s.latestDate,
    isActive: s.isActive,
  }));
  return JSON.stringify({ type: "detected_subscriptions", currency, subscriptions: items }, null, 2);
}

export function buildAffordabilityContext(
  result: AffordabilityResult,
  currency: string,
  itemDescription: string,
  isRecurring: boolean
): string {
  return JSON.stringify({
    type: "affordability_check",
    currency,
    item: itemDescription,
    isRecurring,
    canAfford: result.canAfford,
    monthlySurplus: `${currency} ${toMajor(result.monthlySuplusMinor)}`,
    remainingAfter: result.canAfford ? `${currency} ${toMajor(result.remainingAfterMinor)}` : null,
    shortfall: !result.canAfford ? `${currency} ${toMajor(result.shortfallMinor)}` : null,
    comfortLevel: result.comfortLevel,
  }, null, 2);
}

export function buildHealthScoreContext(result: HealthScoreResult): string {
  return JSON.stringify({
    type: "financial_health_score",
    overallScore: result.score,
    grade: result.grade,
    components: {
      savings: result.components.savingsScore,
      budgetAdherence: result.components.budgetScore,
      emergencyFund: result.components.emergencyFundScore,
      debtManagement: result.components.debtScore,
    },
  }, null, 2);
}
```

### Task 5: Create Narrators

**Files:**
- Create: `ai/narrators.ts`

Each narrator function is async, calls Claude Haiku with the system prompt + context, and returns `NarrativeResult`. All follow the same pattern.

**Step 1: Create ai/narrators.ts**

```typescript
/**
 * Narrator functions — one per insight type.
 *
 * Each narrator:
 * 1. Accepts typed engine output
 * 2. Builds JSON context via context builder
 * 3. Calls Claude Haiku with locked system prompt
 * 4. Returns NarrativeResult
 *
 * No financial math. No DB access. All numbers pre-computed.
 */
import type { SpendingTrend } from "../engine/insights/spending-trends";
import type { SpendingAnomaly } from "../engine/insights/anomaly-detection";
import type { DetectedSubscription } from "../engine/insights/subscription-detection";
import type { AffordabilityResult } from "../engine/insights/affordability";
import type { HealthScoreResult } from "../engine/insights/health-score";
import type { NarrativeResult } from "./types";
import { getAnthropicClient, NARRATIVE_MODEL, NARRATIVE_MAX_TOKENS } from "./client";
import { NARRATIVE_SYSTEM_PROMPT } from "./system-prompt";
import {
  buildSpendingTrendsContext,
  buildAnomalyContext,
  buildSubscriptionContext,
  buildAffordabilityContext,
  buildHealthScoreContext,
} from "./context-builders";

async function callNarrative(userMessage: string): Promise<NarrativeResult> {
  try {
    const client = getAnthropicClient();
    const response = await client.messages.create({
      model: NARRATIVE_MODEL,
      max_tokens: NARRATIVE_MAX_TOKENS,
      system: NARRATIVE_SYSTEM_PROMPT,
      messages: [{ role: "user", content: userMessage }],
    });

    const textBlock = response.content.find((b) => b.type === "text");
    if (!textBlock || textBlock.type !== "text") {
      return { ok: false, error: "No text in LLM response" };
    }

    return { ok: true, narrative: textBlock.text };
  } catch (err: unknown) {
    const message = err instanceof Error ? err.message : "Unknown error";
    return { ok: false, error: message };
  }
}

export async function narrateSpendingTrends(
  trends: readonly SpendingTrend[],
  currency: string
): Promise<NarrativeResult> {
  if (trends.length === 0) {
    return { ok: true, narrative: "No spending data available for trend analysis." };
  }
  const context = buildSpendingTrendsContext(trends, currency);
  return callNarrative(
    `Explain these spending trends in 2-4 sentences. Focus on the most significant changes.\n\n${context}`
  );
}

export async function narrateAnomalies(
  anomalies: readonly SpendingAnomaly[],
  currency: string
): Promise<NarrativeResult> {
  if (anomalies.length === 0) {
    return { ok: true, narrative: "No unusual spending patterns detected this month." };
  }
  const context = buildAnomalyContext(anomalies, currency);
  return callNarrative(
    `Explain these spending anomalies in 2-4 sentences. Note which categories changed significantly.\n\n${context}`
  );
}

export async function narrateSubscriptions(
  subscriptions: readonly DetectedSubscription[],
  currency: string
): Promise<NarrativeResult> {
  if (subscriptions.length === 0) {
    return { ok: true, narrative: "No recurring subscriptions detected." };
  }
  const context = buildSubscriptionContext(subscriptions, currency);
  return callNarrative(
    `Summarize these detected subscriptions in 2-4 sentences. Mention the total count and highlight the largest.\n\n${context}`
  );
}

export async function narrateAffordability(
  result: AffordabilityResult,
  currency: string,
  itemDescription: string,
  isRecurring: boolean
): Promise<NarrativeResult> {
  const context = buildAffordabilityContext(result, currency, itemDescription, isRecurring);
  return callNarrative(
    `Explain this affordability assessment in 2-4 sentences. State whether the item is affordable and the comfort level.\n\n${context}`
  );
}

export async function narrateHealthScore(
  result: HealthScoreResult
): Promise<NarrativeResult> {
  const context = buildHealthScoreContext(result);
  return callNarrative(
    `Explain this financial health score in 2-4 sentences. Mention the grade and which components are strongest and weakest.\n\n${context}`
  );
}
```
